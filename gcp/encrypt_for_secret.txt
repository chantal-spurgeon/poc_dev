For GCP terraform to encrypt secret value in code.

-- For simple password
project_id="<project>"
keyring_name="<keyring>"
key_name="<key>"
keyring_location="<location>"

	echo -n '<password>' | gcloud kms encrypt \
      --project ${project_id}\
      --location ${keyring_location} \
      --keyring ${keyring_name} \
      --key ${key_name} \
      --plaintext-file=-  \
      --ciphertext-file=- | base64

-- Using file text eg RSA private key or json
project_id="<project>"
keyring_name="<keyring>"
key_name="<key>"
keyring_location="<location>"

	echo -n | gcloud kms encrypt \
      --project ${project_id}\
      --location ${keyring_location} \
      --keyring ${keyring_name} \
      --key ${key_name} \
      --plaintext-file=- <filename> \
      --ciphertext-file=- | base64
